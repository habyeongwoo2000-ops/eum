# E9-Bridge — 검색엔진에게 어디를 보고 어디를 건너뛸지 알려 줍니다.
#
# 검색에서 찾아 들어오는 사람에게 쓸모 있는 쪽만 열어 둡니다.
# 로그인·가입·마이페이지는 검색 결과에 떠 봐야 도움이 안 되고,
# 관리자 페이지는 애초에 남에게 보일 이유가 없습니다.
#
# 주의: robots.txt 는 "요청"이지 "잠금장치"가 아닙니다. 여기 적었다고
# 관리자 페이지가 보호되는 것이 아닙니다. 실제 보호는 서버 쪽
# ADMIN_SETUP_KEY 가 합니다.

User-agent: *

# 검색에 뜰 이유가 없는 곳
Disallow: /admin.html
Disallow: /account.html
Disallow: /login.html
Disallow: /signup.html
Disallow: /api/

# 나머지는 모두 열어 둡니다
Allow: /

Sitemap: https://eum-six.vercel.app/sitemap.xml
